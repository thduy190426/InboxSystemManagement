58: function parseMessageDate(message?: Message) {
59:   const value = message?.createdAt || message?.updatedAt
60: 
61:   if (!value) {
62:     return null
63:   }
64: 
65:   const date = new Date(value)
66: 
67:   return Number.isNaN(date.getTime()) ? null : date
68: }

70: function isSameLocalDay(left: Date | null, right: Date | null) {
71:   if (!left || !right) {
72:     return false
73:   }
74: 
75:   return (
76:     left.getFullYear() === right.getFullYear() &&
77:     left.getMonth() === right.getMonth() &&
78:     left.getDate() === right.getDate()
79:   )
80: }

82: function formatDateHeader(date: Date | null, now = new Date()) {
83:   if (!date) {
84:     return ''
85:   }
86: 
87:   const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
88:   const targetDay = new Date(date.getFullYear(), date.getMonth(), date.getDate())
89:   const dayDistance = Math.round((today.getTime() - targetDay.getTime()) / 86400000)
90: 
91:   if (dayDistance === 0) {
92:     return 'Hôm nay'
93:   }
94: 
95:   if (dayDistance === 1) {
96:     return 'Hôm qua'
97:   }
98: 
99:   return new Intl.DateTimeFormat('vi-VN', {
100:     day: '2-digit',
101:     month: '2-digit',
102:     year: 'numeric',
103:   }).format(date)
104: }

106: function formatMessageTime(message: Message) {
107:   if (message.time === 'Bây giờ' || message.time === 'Đang gửi...') {
108:     return message.time
109:   }
110: 
111:   const date = parseMessageDate(message)
112:   
113:   if (date) {
114:     return date.toLocaleTimeString('vi-VN', {
115:       hour: '2-digit',
116:       minute: '2-digit',
117:     })
118:   }
119: 
120:   return message.time
121: }

564:   function rotateImageFile(file: File, rotation: number): Promise<File> {
565:     return new Promise((resolve, reject) => {
566:       const img = new globalThis.Image()
567:       img.onload = () => {
568:         const canvas = document.createElement('canvas')
569:         const ctx = canvas.getContext('2d')
570:         if (!ctx) return reject('No context')
571:         
572:         if (rotation === 90 || rotation === 270) {
573:           canvas.width = img.height
574:           canvas.height = img.width
575:         } else {
576:           canvas.width = img.width
577:           canvas.height = img.height
578:         }
579:         
580:         ctx.translate(canvas.width / 2, canvas.height / 2)
581:         ctx.rotate((rotation * Math.PI) / 180)
582:         ctx.drawImage(img, -img.width / 2, -img.height / 2)
583:         
584:         canvas.toBlob((blob) => {
585:           if (blob) {
586:             resolve(new File([blob], file.name, { type: file.type }))
587:           } else {
588:             reject('Canvas toBlob failed')
589:           }
590:         }, file.type)
591:       }
592:       img.onerror = reject
593:       img.src = URL.createObjectURL(file)
594:     })
595:   }

1119:       const pattern = mentionNames.length

1121:           `(@(?:${mentionNames
1122:             .map((name) => name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))
1123:             .join('|')}))`,
1124:           'giu',
1125:         )
1126:         : null

1128:       if (!pattern) {
1129:         return linkifyText(message.text)
1130:       }
1131: 
1132:       return message.text.split(pattern).map((part, index) => {
1133:         pattern.lastIndex = 0
1134: 
1135:         return pattern.test(part) ? (
1136:           <mark className="mention-highlight" key={`${part}-${index}`}>
1137:             {part}
1138:           </mark>
1139:         ) : (
1140:           linkifyText(part)
1141:         )
1142:       })
1143:     }
1144: 
1145:     const lowerText = message.text.toLocaleLowerCase('vi-VN')
1146:     const matchIndex = lowerText.indexOf(normalizedSearch)
1147: 
1148:     if (matchIndex === -1) {
1149:       return linkifyText(message.text)
1150:     }
1151: 
1152:     const before = message.text.slice(0, matchIndex)
1153:     const match = message.text.slice(matchIndex, matchIndex + normalizedSearch.length)
1154:     const after = message.text.slice(matchIndex + normalizedSearch.length)
1155: 
1156:     return (
1157:       <>
1158:         {linkifyText(before)}
1159:         <mark>{match}</mark>
